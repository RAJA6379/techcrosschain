// Generate a separate Devnet development wallet.
function getDevelopmentWallet() {
  if (fs.existsSync(walletFile)) {
    const secret = JSON.parse(
      fs.readFileSync(walletFile, "utf8")
    );

    return Keypair.fromSecretKey(
      Uint8Array.from(secret)
    );
  }

  const wallet = Keypair.generate();

  fs.writeFileSync(
    walletFile,
    JSON.stringify(Array.from(wallet.secretKey)),
    { mode: 0o600 }
  );

  return wallet;
}

async function main() {
  const phantomAddress = process.argv[2];

  if (!phantomAddress) {
    throw new Error(
      "Provide your Phantom public wallet address."
    );
  }

  const recipient = new PublicKey(phantomAddress);
  const payer = getDevelopmentWallet();

  console.log("Network: Solana Devnet");
  console.log(
    "Development wallet:",
    payer.publicKey.toBase58()
  );

  if (fs.existsSync(outputFile)) {
    console.log(
      "Token output already exists. No new tokens created."
    );
    return;
  }

  const balance = await connection.getBalance(
    payer.publicKey
  );

  console.log(
    "Development wallet balance:",
    balance / LAMPORTS_PER_SOL,
    "SOL"
  );

  if (balance < 0.05 * LAMPORTS_PER_SOL) {
    console.log(
      "\nFund the development wallet with " +
      "about 0.1 Devnet SOL, then run this " +
      "command again."
    );
    return;
  }

  const results = [];

  for (const token of tokens) {
    console.log(`Creating ${token.symbol}...`);

    const mint = await createMint(
      connection,
      payer,
      payer.publicKey,
      null,
      decimals
    );

    const tokenAccount =
      await getOrCreateAssociatedTokenAccount(
        connection,
        payer,
        mint,
        recipient
      );

    const signature = await mintTo(
      connection,
      payer,
      mint,
      tokenAccount.address,
      payer,
      supply
    );

      name: token.name,
      symbol: token.symbol,
      mint: mint.toBase58(),
      decimals,
      supply: "1000000",

      tokenAccount: tokenAccount.address.toBase58(),

const anchor = require("@coral-xyz/anchor");
const assert = require("assert");

const {
  PublicKey,

const {
  TOKEN_PROGRAM_ID,
  createMint,
  getAccount,

describe("MinSwap Solana - Pool Initialization", () => {
  // Use the Anchor provider from Anchor.toml
  const provider = anchor.AnchorProvider.env();
  anchor.setProvider(provider);

  const program = anchor.workspace.minswapSolana;
  const connection = provider.connection;
  const wallet = provider.wallet;

  it("Creates a pool and two token vaults", async () => {
    // STEP 1: Create two temporary Localnet SPL tokens.
    const mint1 = await createMint(
      connection,
      wallet.payer,
      wallet.publicKey,
      null,
      6
    );

    const mint2 = await createMint(
      connection,
      wallet.payer,
      wallet.publicKey,
      null,
      6
    );

    // STEP 2: Sort the mint addresses.
    // Your Rust contract requires token A < token B.
    const [tokenA, tokenB] =
      Buffer.compare(mint1.toBuffer(), mint2.toBuffer()) < 0
        ? [mint1, mint2]
        : [mint2, mint1];

    // STEP 3: Derive the pool PDA.
    const [poolPda] = PublicKey.findProgramAddressSync(
      [
        Buffer.from("pool"),
        tokenA.toBuffer(),
        tokenB.toBuffer(),
      ],
      program.programId
    );

    // STEP 4: Derive the token vault PDAs.
    const [vaultA] = PublicKey.findProgramAddressSync(
      [
        Buffer.from("vault_a"),
        poolPda.toBuffer(),
      ],
      program.programId
    );

    const [vaultB] = PublicKey.findProgramAddressSync(
      [
        Buffer.from("vault_b"),
        poolPda.toBuffer(),
      ],
      program.programId
    );

    // STEP 5: Initialize the pool with every required account.
    const tx = await program.methods
      .initializePool()
      .accountsStrict({
        authority: wallet.publicKey,
        pool: poolPda,
        tokenAMint: tokenA,
        tokenBMint: tokenB,
        tokenAVault: vaultA,
        tokenBVault: vaultB,
        tokenProgram: TOKEN_PROGRAM_ID,
        systemProgram: SystemProgram.programId,
        rent: SYSVAR_RENT_PUBKEY,
      })
      .rpc();

    // STEP 6: Read and verify the pool state.
    const pool = await program.account.pool.fetch(poolPda);

    assert.strictEqual(
      pool.tokenAMint.toBase58(),
      tokenA.toBase58()
    );

    assert.strictEqual(
      pool.tokenBMint.toBase58(),
      tokenB.toBase58()
    );

    assert.strictEqual(
      pool.tokenAVault.toBase58(),
      vaultA.toBase58()
    );

    assert.strictEqual(
      pool.tokenBVault.toBase58(),
      vaultB.toBase58()
    );

    assert.strictEqual(
      pool.authority.toBase58(),
      wallet.publicKey.toBase58()
    );

    assert.strictEqual(pool.totalLiquidity.toString(), "0");
    assert.strictEqual(pool.feeBps, 30);

    // STEP 7: Verify both SPL token vault accounts.
    const tokenAVault = await getAccount(connection, vaultA);
    const tokenBVault = await getAccount(connection, vaultB);

    assert.strictEqual(
      tokenAVault.owner.toBase58(),
      poolPda.toBase58()
    );

    assert.strictEqual(
      tokenBVault.owner.toBase58(),
      poolPda.toBase58()
    );

    assert.strictEqual(tokenAVault.amount, 0n);
    assert.strictEqual(tokenBVault.amount, 0n);

    console.log("Pool initialized successfully!");
    console.log("Pool address:", poolPda.toBase58());
    console.log("Token A vault:", vaultA.toBase58());
    console.log("Token B vault:", vaultB.toBase58());
    console.log("Transaction:", tx);
  });
});

      const d = e.detail;
      if (!d?.info) return;
      setList((prev) => (prev.some((w) => w.info.uuid === d.info.uuid) ? prev : [...prev, d]));

    window.addEventListener("eip6963:announceProvider", onAnnounce);

// Adapts a Wallet Standard Solana account (Phantom / MetaMask Solana, as connected in src/solana/wallet.js)
// into the AnchorProvider that the Garden SDK expects for Solana swaps.
import { Transaction, VersionedTransaction } from "@solana/web3.js";

export async function makeAnchorProvider(connection, rpcConnection) {
  const { AnchorProvider } = await import("@coral-xyz/anchor");
  const sign = connection?.wallet?.features?.["solana:signTransaction"];
  if (!sign) throw new Error(`${connection?.walletName || "This wallet"} does not support signing Solana transactions.`);

  const signOne = async (tx) => {
    const versioned = "version" in tx;
    const bytes = versioned ? tx.serialize() : tx.serialize({ requireAllSignatures: false, verifySignatures: false });
    const [{ signedTransaction }] = await sign.signTransaction({
      account: connection.account,
      transaction: bytes,
      chain: "solana:devnet",
    });
    return versioned ? VersionedTransaction.deserialize(signedTransaction) : Transaction.from(signedTransaction);
  };

  const wallet = {
    publicKey: connection.publicKey,
    signTransaction: signOne,
    signAllTransactions: async (txs) => Promise.all(txs.map(signOne)),
  };
  return new AnchorProvider(rpcConnection, wallet, { commitment: "confirmed" });
}

function getSelectedWallet(walletName) {

    const wallet = metamaskClient?.getWallet();

      throw new Error("MetaMask Solana is not initialized.");

      (wallet) =>
        wallet.name.toLowerCase().includes("phantom") &&
        wallet.chains.some((chain) =>
          chain.startsWith("solana:")
        )
    );

    if (!phantom) {
      throw new Error(

    return phantom;
  }

  const connectFeature =
    wallet.features["standard:connect"];

    throw new Error(
      `${walletName} does not support wallet connection.`
    );

        setStatus("Order created. Track it below.");
      }
      setAmount("");
    } catch (e) {
      setError(e?.shortMessage || e?.message || "Swap failed.");
      setStatus("");
    } finally { setBusy(false); }
  }

  return (
    <section className="swap-card" style={{ width: "100%", maxWidth: 480, margin: "28px auto" }}>
      <div className="card-top"><div><h1>Swap</h1><p>Swap across Sepolia, Solana Devnet and Bitcoin testnet.</p></div></div>

          <TokenButton token={from} onClick={() => setPicker("from")} />
        </div>
      </div>

      <div className="flip-wrap"><button className="flip" onClick={flip} aria-label="Flip tokens" type="button">↓↑</button></div>

          <TokenButton token={to} onClick={() => setPicker("to")} />

        <div className="field">

      )}

      {quote && route.kind === ROUTE.GARDEN && <p className="notice">Fee: {(quote.feeBps / 100).toFixed(2)}% · Route: Garden (native Bitcoin, non-custodial HTLC)</p>}
      {quote && route.kind === ROUTE.LOCAL && <p className="notice">Minimum received (0.5% slippage): {quote.minDisplay} {to.symbol}</p>}
      {route.kind === ROUTE.NONE && <div className="error" role="alert">{route.reason}</div>}
      {needsKey && <div className="error" role="alert">Cross-chain swaps need a Garden API key. Copy .env.example to .env, set VITE_GARDEN_API_KEY, and restart the dev server.</div>}
      {quoteError && <div className="error" role="alert">{quoteError}</div>}

     {solChoice && !solanaAccount && (
  <div className="sol-choice">
    <button
      className="swap-button"
      disabled={solanaBusy}
      onClick={() => {
        setSolChoice(false);
        onConnectSolana("Phantom");
      }}
    >
      Connect Phantom
    </button>
  </div>
)}

      <button className="swap-button" disabled={action.disabled || busy} onClick={action.run}>{busy ? "Working..." : action.label}</button>
      {status && <p className="notice" role="status">{status}</p>}
      {error && <div className="error" role="alert">{error}</div>}
      {lastBtcTx && <p className="notice">BTC transfer sent: <a href={btcExplorerTx(lastBtcTx)} target="_blank" rel="noreferrer">view on mempool.space</a></p>}

      {picker && <TokenPicker onPick={(id) => pick(picker, id)} onClose={() => setPicker(null)} />}
      <OrderStatus orders={orders} />
    </section>
  );
}

import { BrowserProvider, formatEther } from "ethers";

// Discover wallets using EIP-6963.
export async function discoverWallets() {
  const found = new Map();

  const listener = (event) => {
    const { info, provider } = event.detail || {};

    if (info?.uuid && provider?.request) {
      found.set(info.uuid, {
        id: info.uuid,
        name: info.name,
        icon: info.icon,
        provider,
      });
    }
  };

  window.addEventListener(
    "eip6963:announceProvider",
    listener
  );

  window.dispatchEvent(
    new Event("eip6963:requestProvider")
  );

  await new Promise((resolve) =>
    setTimeout(resolve, 350)
  );

  window.removeEventListener(
    "eip6963:announceProvider",
    listener
  );

  const wallets = [...found.values()];

  // Support older extensions as a fallback.
  const injected = window.ethereum;

  const legacy = Array.isArray(injected?.providers)
    ? injected.providers
    : injected
      ? [injected]
      : [];

  // Only use legacy fallback if EIP-6963 found none.
  for (const provider of (wallets.length ? [] : legacy)) {
    if (
      !provider?.request ||
      wallets.some(
        (wallet) => wallet.provider === provider
      )
    ) {
      continue;
    }

    const name = provider.isTrust
      ? "Trust Wallet"
      : provider.isMetaMask && !provider.isTrust
        ? "MetaMask (legacy)"
        : "Injected wallet";

    wallets.push({
      id: `legacy-${wallets.length}`,
      name,
      icon: null,
      provider,
    });
  }

  return wallets;
}

// Connect to the exact wallet selected by the user.
export async function connectWallet(wallet) {
  if (!wallet?.provider?.request) {
    throw new Error("Select an available wallet.");
  }

  try {
    const accounts = await wallet.provider.request({
      method: "eth_requestAccounts",
    });

    if (!accounts?.[0]) {
      throw new Error(
        "No wallet account was selected."
      );
    }

    const provider = new BrowserProvider(
      wallet.provider
    );

    const network = await provider.getNetwork();

    const balance = await provider.getBalance(
      accounts[0]
    );

    return {
      provider,
      injectedProvider: wallet.provider,
      walletName: wallet.name,
      address: accounts[0],
      chainId: Number(network.chainId),
      balance: formatEther(balance),
    };
  } catch (error) {
    if (
      error?.code === 4001 ||
      error?.info?.error?.code === 4001
    ) {
      throw new Error(
        "Wallet connection was rejected."
      );
    }

    throw new Error(
      error?.shortMessage ||
      error?.info?.error?.message ||
      error?.message ||
      "Unable to connect wallet. Unlock or restart the wallet extension and try again."
    );
  }
}

// Short display address.
export function shortAddress(address) {
  return address
    ? `${address.slice(0, 6)}...${address.slice(-4)}`
    : "";
}
